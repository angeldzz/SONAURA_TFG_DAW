----permissions.py:
cuando tenemos dos clases del modelo con distinto nombre en la relacion a User, hacemos dos classes IsOwnerOrReadOnly

from rest_framework import permissions

class IsOwnerOrReadOnly(permissions.BasePermission):
    def has_permission(self, request, view):
        return request.user.is_authenticated

    def has_object_permission(self, request, view, obj):
        # Los permisos de lectura están permitidos para cualquier solicitud,
        # por lo que siempre permitiremos solicitudes GET, HEAD o OPTIONS.
        if request.method in permissions.SAFE_METHODS:
            return True

        # Los permisos de escritura solo están permitidos para el propietario del objeto.
        return obj.owner == request.user

views.py de api

from rest_framework import viewsets
from rest_framework.permissions import IsAuthenticated
from base.models import Categoria,Productos
from .serializers import CategoriaSerializer,ProductosSerializer
from .permissions import IsOwnerOrReadOnly

class CategoriaViewSet(viewsets.ModelViewSet):
    """
    API que permite:
    - Ver todos los objetos (solo usuarios autenticados)
    - Crear un objeto (se asigna automáticamente al usuario autenticado)
    - Modificar y eliminar solo si el usuario es el creador
    """
    
    queryset = Categoria.objects.all()
    serializer_class = CategoriaSerializer
    permission_classes = [IsAuthenticated,IsOwnerOrReadOnly]

    def perform_create(self, serializer):
        """Asigna automáticamente el usuario autenticado como creador"""
        serializer.save(creador=self.request.user)    


class ProductosViewSet(viewsets.ModelViewSet):

Para optener los objetos:
curl -X GET http://127.0.0.1:8000/api/mis_objetos/ ^
-H "Authorization: Bearer <token access>"

Para crear un objeto:
curl -X POST http://127.0.0.1:8000/api/mis_objetos/ ^
-H "Authorization: Bearer <token access>" ^
-H "Content-type: application/json" ^
-d "{\"campo1\":\"esto es el texto 1\",\"campo2\":\"esto es el texto 2\"}"

Para modificar:
curl -X PUT http://127.0.0.1:8000/api/mis_objetos/5/ ^
-H "Authorization: Bearer <token access>" ^
-H "Content-type: application/json" ^
-d "{\"campo1\":\"modificado 1\",\"campo2\":\"modificado 2\"}"

Para eliminar:
curl -X DELETE http://127.0.0.1:8000/api/mis_objetos/1/ ^
-H "Authorization: Bearer <token access>"
